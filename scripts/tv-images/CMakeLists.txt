cmake_minimum_required(VERSION 3.16)
project(LeanTubeTVImages C)

add_library(SDL3::Headers INTERFACE IMPORTED)
set_target_properties(SDL3::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${TV_INCLUDE}")
add_library(SDL3::SDL3-shared SHARED IMPORTED)
set_target_properties(SDL3::SDL3-shared PROPERTIES IMPORTED_LOCATION "${TV_RUNTIME}/libSDL3.so.0" INTERFACE_LINK_LIBRARIES SDL3::Headers)

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SDLIMAGE_VENDORED ON CACHE BOOL "" FORCE)
set(SDLIMAGE_WEBP_SHARED OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_WEBP_SAVE OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_INSTALL OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_BACKEND_STB OFF CACHE BOOL "" FORCE)
foreach(format ANI AVIF BMP GIF JPG JXL LBM PCX PNG PNM QOI TGA TIF XCF XPM XV)
	set(SDLIMAGE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDLIMAGE_SVG ON CACHE BOOL "" FORCE)
set(SDLIMAGE_WEBP ON CACHE BOOL "" FORCE)
foreach(tool ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO WEBPMUX EXTRAS WEBP_JS)
	set(WEBP_BUILD_${tool} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${IMAGE_SOURCE}" SDL_image)
